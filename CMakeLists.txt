# CMakeLists.txt for ArmorDetector

target_include_directories(xr PUBLIC ${CMAKE_CURRENT_LIST_DIR})

get_filename_component(_MODULE_DIRNAME ${CMAKE_CURRENT_LIST_DIR} NAME)
set(_DEPS_TARGET "${_MODULE_DIRNAME}_deps")
add_library(${_DEPS_TARGET} INTERFACE)

find_package(OpenCV REQUIRED)
find_package(OpenVINO QUIET COMPONENTS Runtime)
find_package(HailoRT QUIET CONFIG)

if(NOT OpenVINO_FOUND AND NOT HailoRT_FOUND)
  message(FATAL_ERROR "ArmorDetector requires either OpenVINO or HailoRT")
endif()

remove_definitions(-D_GLIBCXX_USE_CXX11_ABI=0)
target_compile_definitions(xr PUBLIC _GLIBCXX_USE_CXX11_ABI=1)

target_link_libraries(${_DEPS_TARGET} INTERFACE
  ${OpenCV_LIBS}
)

if(OpenVINO_FOUND)
  get_target_property(OPENVINO_RUNTIME_INCLUDE_DIRS openvino::runtime INTERFACE_INCLUDE_DIRECTORIES)
  get_target_property(OPENVINO_RUNTIME_LIBRARY openvino::runtime IMPORTED_LOCATION_RELEASE)
  if(NOT OPENVINO_RUNTIME_LIBRARY)
    get_target_property(OPENVINO_RUNTIME_LIBRARY openvino::runtime IMPORTED_LOCATION)
  endif()
  if(NOT OPENVINO_RUNTIME_LIBRARY)
    message(FATAL_ERROR "Failed to resolve openvino::runtime library path")
  endif()

  target_include_directories(${_DEPS_TARGET} INTERFACE
    ${OPENVINO_RUNTIME_INCLUDE_DIRS}
  )
  target_link_libraries(${_DEPS_TARGET} INTERFACE
    ${OPENVINO_RUNTIME_LIBRARY}
  )
endif()

if(HailoRT_FOUND)
  target_link_libraries(${_DEPS_TARGET} INTERFACE
    HailoRT::libhailort
  )
endif()

set(ARMOR_DETECTOR_MODEL_FILE
  "${CMAKE_CURRENT_LIST_DIR}/model/armor_detector_640x512.onnx"
)
if(OpenVINO_FOUND AND NOT EXISTS "${ARMOR_DETECTOR_MODEL_FILE}")
  message(FATAL_ERROR "ArmorDetector model file not found: ${ARMOR_DETECTOR_MODEL_FILE}")
endif()

set(ARMOR_DETECTOR_SKD_MODEL_FILE
  "${CMAKE_CURRENT_LIST_DIR}/model/SKD250526.onnx"
)

set(ARMOR_DETECTOR_SKD_INT8_HEAD_L_HEF_FILE
  "${CMAKE_CURRENT_LIST_DIR}/model/skd_int8_head_l.hef"
)
set(ARMOR_DETECTOR_SKD_INT8_GRID_L_HEF_FILE
  "${CMAKE_CURRENT_LIST_DIR}/model/skd_int8_grid_l.hef"
)
set(ARMOR_DETECTOR_SZU_INT16_HEAD_L_HEF_FILE
  "${CMAKE_CURRENT_LIST_DIR}/model/szu_int16_head_l.hef"
)
set(ARMOR_DETECTOR_SKD_INT8_HEAD_HEF_FILE
  "${CMAKE_CURRENT_LIST_DIR}/model/skd_int8_head.hef"
)
set(ARMOR_DETECTOR_SKD_INT8_GRID_HEF_FILE
  "${CMAKE_CURRENT_LIST_DIR}/model/skd_int8_grid.hef"
)
set(ARMOR_DETECTOR_SZU_INT16_HEAD_HEF_FILE
  "${CMAKE_CURRENT_LIST_DIR}/model/szu_int16_head.hef"
)
set(ARMOR_DETECTOR_SZU_TAIL_ONNX_FILE
  "${CMAKE_CURRENT_LIST_DIR}/model/szu_tail.onnx"
)

target_compile_definitions(${_DEPS_TARGET} INTERFACE
  ARMOR_DETECTOR_MODEL_PATH="${ARMOR_DETECTOR_MODEL_FILE}"
  ARMOR_DETECTOR_SKD_MODEL_PATH="${ARMOR_DETECTOR_SKD_MODEL_FILE}"
  ARMOR_DETECTOR_SKD_INT8_HEAD_L_HEF_PATH="${ARMOR_DETECTOR_SKD_INT8_HEAD_L_HEF_FILE}"
  ARMOR_DETECTOR_SKD_INT8_GRID_L_HEF_PATH="${ARMOR_DETECTOR_SKD_INT8_GRID_L_HEF_FILE}"
  ARMOR_DETECTOR_SZU_INT16_HEAD_L_HEF_PATH="${ARMOR_DETECTOR_SZU_INT16_HEAD_L_HEF_FILE}"
  ARMOR_DETECTOR_SKD_INT8_HEAD_HEF_PATH="${ARMOR_DETECTOR_SKD_INT8_HEAD_HEF_FILE}"
  ARMOR_DETECTOR_SKD_INT8_GRID_HEF_PATH="${ARMOR_DETECTOR_SKD_INT8_GRID_HEF_FILE}"
  ARMOR_DETECTOR_SZU_INT16_HEAD_HEF_PATH="${ARMOR_DETECTOR_SZU_INT16_HEAD_HEF_FILE}"
  ARMOR_DETECTOR_SZU_TAIL_ONNX_PATH="${ARMOR_DETECTOR_SZU_TAIL_ONNX_FILE}"
)

if(OpenVINO_FOUND)
  target_compile_definitions(${_DEPS_TARGET} INTERFACE
    OV_THREAD=OV_THREAD_TBB
    ARMOR_DETECTOR_HAVE_OPENVINO=1
  )
endif()

if(HailoRT_FOUND)
  target_compile_definitions(${_DEPS_TARGET} INTERFACE
    ARMOR_DETECTOR_HAVE_HAILORT=1
  )
endif()

set_property(GLOBAL APPEND PROPERTY XR_MODULE_DEPS ${_DEPS_TARGET})
